/**
 * QRコード受付システム - Google Apps Script

// シート名
const SHEET = {
  SETTINGS: '設定',
  PARTICIPANTS: '参加者リスト',
};

/**
 * 設定シートから値を取得
 */
function getSettings() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET.SETTINGS);
  if (!sheet) return null;

  const data = sheet.getRange('A2:B10').getValues();
  const settings = {};
  data.forEach(row => {
    if (row[0]) settings[row[0]] = row[1];
  });
  return settings;
}

/**
 * スプレッドシートを初期化（初回のみ実行）
 */
function initializeSpreadsheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // 設定シート作成
  let settingsSheet = ss.getSheetByName(SHEET.SETTINGS);
  if (!settingsSheet) {
    settingsSheet = ss.insertSheet(SHEET.SETTINGS);
  }
  settingsSheet.clear();
  settingsSheet.getRange('A1:B1').setValues([['項目', '値']]);
  settingsSheet.getRange('A1:B1').setFontWeight('bold').setBackground('#6aa84f').setFontColor('white');
  settingsSheet.getRange('A2:B8').setValues([
    ['イベント名', 'サンプルイベント'],
    ['イベント日時', '2025年2月1日（土）13:00〜'],
    ['会場', '東京都渋谷区○○ビル 3F'],
    ['メール件名', '【{イベント名}】受付用QRコードのご案内'],
    ['メール本文（冒頭）', 'この度はお申し込みいただきありがとうございます。'],
    ['メール本文（末尾）', '当日お会いできることを楽しみにしております。'],
    ['送信者名', ''],
  ]);

  // 参加者リストシート作成
  let participantsSheet = ss.getSheetByName(SHEET.PARTICIPANTS);
  if (!participantsSheet) {
    participantsSheet = ss.insertSheet(SHEET.PARTICIPANTS);
  }
  participantsSheet.clear();
  participantsSheet.getRange('A1:H1').setValues([[
    '受付コード', '名前', 'メールアドレス', '所属', 'メール送信', '送信日時', '受付状況', '受付日時'
  ]]);
  participantsSheet.getRange('A1:H1').setFontWeight('bold').setBackground('#4a86e8').setFontColor('white');
  participantsSheet.setFrozenRows(1);

  // サンプルデータ
  participantsSheet.getRange('A2:D4').setValues([
    [generateCode(), '山田太郎', 'yamada@example.com', '営業部'],
    [generateCode(), '鈴木花子', 'suzuki@example.com', '開発部'],
    [generateCode(), '田中一郎', 'tanaka@example.com', '人事部'],
  ]);

  SpreadsheetApp.getUi().alert('初期化完了！');
}

/**
 * ユニークな受付コードを生成
 */
function generateCode() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let code = '';
  for (let i = 0; i < 8; i++) {
    code += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return code;
}

/**
 * 選択した行にコードを生成
 */
function generateCodesForSelected() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET.PARTICIPANTS);
  const range = sheet.getActiveRange();
  const startRow = range.getRow();
  const numRows = range.getNumRows();
  let count = 0;

  for (let i = 0; i < numRows; i++) {
    const row = startRow + i;
    if (row === 1) continue;
    const codeCell = sheet.getRange(row, 1);
    if (!codeCell.getValue()) {
      codeCell.setValue(generateCode());
      count++;
    }
  }
  SpreadsheetApp.getUi().alert(`${count}件の受付コードを生成しました`);
}

/**
 * 全員にQRコード付きメールを送信
 */
function sendQREmailToAll() {
  const ui = SpreadsheetApp.getUi();
  const settings = getSettings();
  if (!settings) {
    ui.alert('エラー', '設定シートが見つかりません。', ui.ButtonSet.OK);
    return;
  }

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET.PARTICIPANTS);
  const data = sheet.getDataRange().getValues();

  let unsent = 0;
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] && data[i][2] && data[i][4] !== '済') unsent++;
  }

  if (unsent === 0) {
    ui.alert('全員に送信済みです');
    return;
  }

  const response = ui.alert('確認', `未送信の${unsent}名にメールを送信しますか？`, ui.ButtonSet.YES_NO);
  if (response !== ui.Button.YES) return;

  let successCount = 0;
  for (let i = 1; i < data.length; i++) {
    const row = i + 1;
    const code = data[i][0];
    const name = data[i][1];
    const email = data[i][2];
    if (!code || !email || data[i][4] === '済') continue;

    try {
      sendQREmail(email, name, code, settings);
      sheet.getRange(row, 5).setValue('済');
      sheet.getRange(row, 6).setValue(new Date());
      sheet.getRange(row, 5, 1, 2).setBackground('#e8f5e9');
      successCount++;
      Utilities.sleep(500);
    } catch (e) {
      sheet.getRange(row, 5).setValue('失敗');
      sheet.getRange(row, 6).setValue(e.message);
    }
  }
  ui.alert(`送信完了！成功: ${successCount}件`);
}

/**
 * QRコード付きメールを送信
 */
function sendQREmail(email, name, code, settings) {
  const qrUrl = `https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=${code}`;
  const qrBlob = UrlFetchApp.fetch(qrUrl).getBlob().setName('qrcode.png');

  const eventName = settings['イベント名'] || 'イベント';
  const subject = (settings['メール件名'] || '【{イベント名}】受付用QRコード').replace('{イベント名}', eventName);

  const htmlBody = `
    <div style="font-family: sans-serif; max-width: 600px;">
      <h2 style="color: #4a86e8;">${eventName}</h2>
      <p>${name} 様</p>
      <p>${settings['メール本文（冒頭）'] || ''}</p>
      <div style="background: #f5f5f5; padding: 20px; border-radius: 8px; text-align: center; margin: 20px 0;">
        <img src="cid:qrcode" alt="QRコード" style="width: 250px;"><br>
        <p style="font-size: 24px; font-weight: bold; letter-spacing: 4px;">${code}</p>
      </div>
      <p>${settings['メール本文（末尾）'] || ''}</p>
    </div>
  `;

  GmailApp.sendEmail(email, subject, '', {
    htmlBody: htmlBody,
    inlineImages: { qrcode: qrBlob }
  });
}

/**
 * 受付処理
 */
function processCheckin(code, method) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET.PARTICIPANTS);
  const data = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === code) {
      const row = i + 1;
      const name = data[i][1];

      if (data[i][6] === '済') {
        return { success: false, message: `${name}さんは既に受付済みです`, alreadyCheckedIn: true };
      }

      const now = new Date();
      sheet.getRange(row, 7).setValue('済');
      sheet.getRange(row, 8).setValue(now);
      sheet.getRange(row, 7, 1, 2).setBackground('#c8e6c9');

      return {
        success: true,
        message: `${name}さんの受付完了！`,
        name: name,
        time: Utilities.formatDate(now, 'Asia/Tokyo', 'HH:mm:ss')
      };
    }
  }
  return { success: false, message: `コード「${code}」が見つかりません`, notFound: true };
}

/**
 * 受付状況を取得
 */
function getReceptionStatus() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET.PARTICIPANTS);
  const data = sheet.getDataRange().getValues();
  let total = 0, checkedIn = 0;

  for (let i = 1; i < data.length; i++) {
    if (data[i][0]) {
      total++;
      if (data[i][6] === '済') checkedIn++;
    }
  }
  return { total, checkedIn, remaining: total - checkedIn };
}

/**
 * 参加者を検索
 */
function searchParticipant(query) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET.PARTICIPANTS);
  const data = sheet.getDataRange().getValues();
  const results = [];
  const q = query.toLowerCase();

  for (let i = 1; i < data.length; i++) {
    const code = String(data[i][0]).toLowerCase();
    const name = String(data[i][1]).toLowerCase();
    if (code.includes(q) || name.includes(q)) {
      results.push({
        id: data[i][0],
        name: data[i][1],
        organization: data[i][3],
        checkedIn: data[i][6] === '済'
      });
    }
  }
  return results;
}

    .addToUi();
}

function openReceptionPage() {
  const url = ScriptApp.getService().getUrl();
  if (!url) {
    SpreadsheetApp.getUi().alert('まずWebアプリとしてデプロイしてください');
    return;
  }
  const html = HtmlService.createHtmlOutput(`<script>window.open("${url}");google.script.host.close();</script>`);
  SpreadsheetApp.getUi().showModalDialog(html, '開いています...');
}

  <script>
    let html5QrcodeScanner = null;
    let isProcessing = false;
    let selectedId = null;

    document.addEventListener('DOMContentLoaded', function() {
      updateStatus();
      initQRScanner();
      setInterval(updateStatus, 30000);
    });

    function initQRScanner() {
      html5QrcodeScanner = new Html5Qrcode("qr-reader");
      html5QrcodeScanner.start(
        { facingMode: "environment" },
        { fps: 10, qrbox: { width: 250, height: 250 } },
        onScanSuccess,
        () => {}
      ).catch(err => {
        document.getElementById('qr-section').innerHTML =
          '<p style="color: #f44336; text-align: center; padding: 20px;">カメラを使用できません。手動入力をご利用ください。</p>';
      });
    }

    function onScanSuccess(decodedText) {
      if (isProcessing) return;
      isProcessing = true;
      html5QrcodeScanner.pause();
      processCheckin(decodedText, 'QR');
    }

    function switchTab(tab) {
      const tabs = document.querySelectorAll('.tab');
      tabs.forEach(t => t.classList.remove('active'));
      if (tab === 'qr') {
        tabs[0].classList.add('active');
        document.getElementById('qr-section').classList.remove('hidden');
        document.getElementById('manual-section').classList.remove('active');
      } else {
        tabs[1].classList.add('active');
        document.getElementById('qr-section').classList.add('hidden');
        document.getElementById('manual-section').classList.add('active');
      }
    }

    function handleSearch(query) {
      const resultsDiv = document.getElementById('search-results');
      if (query.length < 1) { resultsDiv.classList.add('hidden'); return; }

      google.script.run
        .withSuccessHandler(function(results) {
          if (results.length === 0) {
            resultsDiv.innerHTML = '<p style="color: #999; padding: 12px;">該当なし</p>';
          } else {
            resultsDiv.innerHTML = results.map(p => `
              <div class="search-item" onclick="selectParticipant('${p.id}', '${p.name}')">
                <div><strong>${p.name}</strong><br><small>${p.id}</small></div>
                <span class="badge ${p.checkedIn ? 'badge-success' : 'badge-pending'}">
                  ${p.checkedIn ? '受付済' : '未受付'}
                </span>
              </div>
            `).join('');
          }
          resultsDiv.classList.remove('hidden');
        })
        .searchParticipant(query);
    }

    function selectParticipant(id, name) {
      selectedId = id;
      document.getElementById('search-input').value = `${name} (${id})`;
      document.getElementById('search-results').classList.add('hidden');
    }

    function manualCheckin() {
      const input = document.getElementById('search-input').value.trim();
      if (!input) { alert('IDまたは名前を入力してください'); return; }
      processCheckin(selectedId || input, '手動');
    }

    function processCheckin(id, method) {
      google.script.run
        .withSuccessHandler(handleResult)
        .withFailureHandler(err => { alert('エラー: ' + err); isProcessing = false; resetScanner(); })
        .processCheckin(id, method);
    }

    function handleResult(response) {
      const resultDiv = document.getElementById('result');
      const iconDiv = document.getElementById('result-icon');
      const messageDiv = document.getElementById('result-message');
      const detailDiv = document.getElementById('result-detail');

      if (response.success) {
        resultDiv.className = 'result show result-success';
        iconDiv.textContent = '✓';
        messageDiv.textContent = response.message;
        detailDiv.textContent = `受付時刻: ${response.time}`;
        updateStatus();
      } else if (response.alreadyCheckedIn) {
        resultDiv.className = 'result show result-warning';
        iconDiv.textContent = '⚠';
        messageDiv.textContent = response.message;
        detailDiv.textContent = '';
      } else {
        resultDiv.className = 'result show result-error';
        iconDiv.textContent = '✗';
        messageDiv.textContent = response.message;
        detailDiv.textContent = 'IDを確認してください';
      }

      document.getElementById('qr-section').classList.add('hidden');
      document.getElementById('manual-section').classList.remove('active');
      isProcessing = false;
      selectedId = null;
    }

    function resetScanner() {
      document.getElementById('result').classList.remove('show');
      document.getElementById('qr-section').classList.remove('hidden');
      document.getElementById('search-input').value = '';
      document.getElementById('search-results').classList.add('hidden');

      const activeTab = document.querySelector('.tab.active');
      if (activeTab.textContent.includes('QR') && html5QrcodeScanner) {
        html5QrcodeScanner.resume();
      } else {
        document.getElementById('manual-section').classList.add('active');
      }
    }

    function updateStatus() {
      google.script.run
        .withSuccessHandler(function(status) {
          document.getElementById('total-count').textContent = status.total;
          document.getElementById('checkedin-count').textContent = status.checkedIn;
          document.getElementById('remaining-count').textContent = status.remaining;
        })
        .getReceptionStatus();

使用しているAPIとサービスの詳細情報

qr_code_2

QR Code Generator API

本システムでは、goQR.me が提供する無料のQRコード生成APIを使用しています。

link API エンドポイント

https://api.qrserver.com/v1/create-qr-code/

check_circle 主な特徴

  • ✓ 完全無料：商用利用も可能
  • ✓ 登録不要：APIキー不要で即利用可
  • ✓ 高可用性：安定したサービス提供
  • ✓ カスタマイズ：サイズ・色・余白など調整可

settings 使用パラメータ

  • size：画像サイズ（例: 250x250）
  • data：埋め込むデータ（受付コード）
  • format：画像形式（png, jpg, svg等）
  • margin：余白サイズ（デフォルト: 1）

code 本システムでの使用例

https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=ABC12345

受付コード「ABC12345」を含む250×250ピクセルのQRコード画像を生成します。

info 公式ドキュメント： https://goqr.me/api/ - 詳細なパラメータや高度な使用方法はこちらを参照してください。

email

Gmail 送信上限（Quota）

Google Apps ScriptからGmailApp.sendEmail()を使用してメールを送信する場合、Googleアカウントの種類によって1日あたりの送信上限が異なります。

アカウント種別 1日あたりの送信上限 備考
person 無料の個人アカウント
（@gmail.com）
100件/日 小規模イベント向け
business Google Workspace
（旧G Suite / 有料プラン）
1,500件/日 大規模イベント・企業利用向け
school Google Workspace for Education
（教育機関向け）
1,500件/日 学校・大学でのイベント向け

warning 上限に達した場合

  • • 24時間後に上限がリセットされます
  • • 上限超過分は「失敗」として記録されます
  • • 翌日に未送信分を再送信してください

tips_and_updates 大量送信のコツ

  • • 数日前から分割して送信する
  • • 本システムは送信済みをスキップします
  • • 100名超の場合はWorkspaceを検討

info 公式ドキュメント： Google Apps Script Quotas - 最新の上限情報はこちらで確認できます。上限は予告なく変更される場合があります。
